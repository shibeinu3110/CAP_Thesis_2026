model EnrollmentSystem

-- Định nghĩa lớp Course (Khóa học)
class Course
attributes
    name : String
end

-- Định nghĩa lớp Student (Sinh viên)
class Student
attributes
    name : String
end

-- Định nghĩa lớp Enrolment (Đăng ký) làm lớp trung gian
class Enrolment
attributes
    grade : Real
end

-- Thiết lập mối quan hệ
association Participation between
    Student[1] role student
    Enrolment[*] role enrolments
end

association Subject between
    Enrolment[*] role enrolments
    Course[1] role course
end

-- Ràng buộc OCL
constraints
context Student
    inv UniqueEnrolmentPerCourse:
        self.enrolments->isUnique(e | e.course)
